{Ramen mit Yuba und Erdnusssauce}

%\graph{% pictures
%    big=pic/schnellekueche/03_ramen-yuba-erdnusssauce
%}

\introduction{%
Ein extrem schnelles Nudelgericht mit cremiger Erdnusssauce und viel Protein durch Yuba (Tofuhaut). In unter 10 Minuten fertig und perfekt für Tage, an denen man einfach nur schnell etwas Warmes will.
}

\ingredients{
    75 g & Ramen Nudeln\index{Getreide!Nudel}\\
    55 g & Yuba / Tofuhaut\index{Tofu \& Fleischalternativen!Yuba}\\
    1 EL & Erdnussbutter\index{Nüsse \& Samen!Erdnuss}\\
    3 EL & Sojasauce\\
    1 EL & Zitronensaft
}

\preparation{%
    \step%
    Ramen nach Packungsanweisung kochen (ca. 3--5 Minuten).
    
    \step%
    Zwei Minuten vor Ende der Garzeit die Yuba-Stangen zu den Nudeln geben und mitkochen.
    
    \step%
    Nudeln und Yuba abgießen und direkt wieder in den Topf geben.
    
    \step%
    Erdnussbutter, Sojasauce und Zitronensaft zugeben und bei niedriger Hitze etwa eine Minute verrühren, bis eine cremige Sauce entsteht.
}

\suggestion[Topping]{%
    Mit 1--2 TL fermentierten Sojabohnen in Chiliöl servieren oder für etwas Crunch mit geröstetem Sesam toppen.
}

\end{recipe}
