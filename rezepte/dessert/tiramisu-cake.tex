\ingredients{
    \textbf{Kuchen} & \\
    90g & Mehl\index{Getreide, Pasta \& Brot!Weizenmehl}\\
    60g & Kakaopulver\index{Backzutaten \& Süßes!Kakaopulver}\\
    200g & Zucker\index{Backzutaten \& Süßes!Zucker}\\
    1 TL & Natron\index{Backzutaten \& Süßes!Natron}\\
    \nicefrac{1}{2} TL & Backpulver\index{Backzutaten \& Süßes!Backpulver}\\
    \nicefrac{1}{2} TL & Salz\index{Kräuter \& Gewürze!Salz}\\
    60ml & Öl\index{Öle, Saucen \& Würzmittel!Pflanzenöl}\\
    120ml & Buttermilch\index{Milchprodukte \& Alternativen!Buttermilch}\\
    1 & Ei\index{Eier!Eier}\\
    1 TL & Vanille\index{Backzutaten \& Süßes!Vanille}\\
    120ml & Warmes Wasser oder Kaffee\index{Getränke!Kaffee}\index{Getränke!Wasser}\\
    \textbf{Creme} & \\
    500g & Mascarpone\index{Milchprodukte \& Alternativen!Mascarpone}\\
    480ml & Sahne\index{Milchprodukte \& Alternativen!Sahne}\\
    65g & Zucker\index{Backzutaten \& Süßes!Zucker}\\
    \nicefrac{1}{4} TL & Salz\index{Kräuter \& Gewürze!Salz}\\
    1 EL & Vanille\index{Backzutaten \& Süßes!Vanille}\\
    2 EL & Instant Espresso\index{Getränke!Espresso}\\
    3 EL & Frisch gemahlener Kaffee\index{Getränke!Kaffee}\\
    \textbf{Topping} & \\
    3 EL & Kakaopulver\index{Backzutaten \& Süßes!Kakaopulver}
}

\preparation{
    \step%
    Backofen auf 175°C vorheizen. 23cm Springform mit Backpapier auslegen.
    
    \step%
    Mehl, Kakao, Zucker, Natron, Backpulver und Salz vermengen.
    
    \step%
    Öl, Buttermilch, Ei, Vanille und Wasser zugeben und glatt rühren.
    
    \step%
    Teig in die Form füllen und 45min backen. Vollständig auskühlen lassen.
    
    \step%
    Für die Creme die Mascarpone und Sahne 30s verrühren.
    
    \step%
    Zucker, Salz, Vanille und Espresso zugeben und bis zu soft peaks schlagen.
    
    \step%
    Gemahlenen Kaffee vorsichtig unterheben.\\
    
    \step%
    Creme auf den ausgekühlten Kuchen streichen und mindestens 4h (besser über Nacht) kühlen.
    
    \step%
    Mit Kakaopulver bestäuben.
}

\hint{Creme nicht übermixen, Mascarpone trennt sich sonst und wird körnig.}

\end{recipe}
