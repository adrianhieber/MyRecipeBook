{Vegane Burger Patties}

%\graph{% pictures
%    big=pic/basics/06_vegane-burger-patties
%}

\introduction{%
Einfache, proteinreiche Burger Patties aus Kidneybohnen und Seitan. Schnell gemacht und perfekt als Basis für Burger oder Bowls.
}

\ingredients{
    1{,}5 Dosen & Kidneybohnen (abgetropft)\index{Hülsenfrüchte!Kidneybohnen}\\
    40 g & Seitan\index{Tofu \& Fleischalternativen!Seitan}\\
    40 g & Haferflocken\index{Getreide, Pasta \& Brot!Haferflocken}\\
    30 g & Mehl\index{Getreide, Pasta \& Brot!Weizenmehl}\\
    2 EL & Senf\index{Öle, Saucen \& Würzmittel!Senf}\\
    1 EL & Gemüsebrühepulver\index{Vorrat \& Konserven!Gemüsebrühe}\\
    1 EL & Sojasauce\index{Öle, Saucen \& Würzmittel!Sojasauce}\\
    3 TL & Paprikapulver\index{Kräuter \& Gewürze!Paprikapulver}\\
    1 TL & Kreuzkümmel\index{Kräuter \& Gewürze!Kreuzkümmel}\\
    \nicefrac{1}{4} TL & Cayennepfeffer\index{Kräuter \& Gewürze!Cayennepfeffer}
}

\preparation{%
    \step%
    Kidneybohnen abtropfen lassen (nicht abspülen) und grob zerstampfen.
    
    \step%
    Mit allen restlichen Zutaten vermengen, bis eine formbare Masse entsteht.
    
    \step%
    Aus der Masse Kugeln formen und flach drücken.
    
    \step%
    In einer Pfanne mit etwas Öl von beiden Seiten goldbraun anbraten.
}

\hint{
    Wenn die Masse zu feucht ist, etwas mehr Haferflocken oder Mehl hinzufügen.
}

\end{recipe}
