{Garlicky Soy Sauce Tofu Bites}

%\graph{% pictures
%    big=pic/hauptgericht/garlic-soy-tofu-bites
%}

\introduction{%
    Knusprige Tofuwürfel in Soja-Sauce. Außen crunchy, innen weich. Am besten genug trinken, da man durch den hohen Salzgehalt der Sojasauce durstg wird.
}

\ingredients{
    \textbf{Tofu} & \\
    300 g & Fester Tofu\index{Protein!Tofu}\\
    1 TL & Salz\\
    1 EL & Knoblauchpulver\\
    \\
    \textbf{Teig} & \\
    120 g & Weizenmehl\\
    60 g & Speisestärke\\
    2 TL & Chilipulver\\
    1 TL & Pfeffer\\
    1 TL & Ital. Kräuter\\
    1 EL & Knoblauchpulver\\
    \nicefrac{1}{2} TL & Zwiebelgranulat\\
    125 ml & Wasser\\
    & Öl zum Frittieren\\
    \\
    \textbf{Sauce} & \\
    3 EL & Vegane Butter\\
    5 Zehen & Knoblauch\index{Geschmacksträger!Knoblauch}\\
    60 ml & Sojasauce\index{Geschmacksträger!Sojasauce}\\\\
    2 EL & Ahornsirup\\
    optional & Sriracha
}

\preparation{%
    \step%
    \textbf{Tofu vorbereiten:} Tofu pressen, in mundgerechte Würfel schneiden und mit Salz und Knoblauchpulver vermengen.
    
    \step%
    \textbf{Panade vorbereiten:} Mehl, Speisestärke, Gewürze und Kräuter mischen. Die Hälfte der Mischung mit 125 ml Wasser zu einem dickflüssigen Teig verrühren.
    
    \step%
    \textbf{Tofu panieren:} Tofuwürfel zuerst im nassen Teig wenden, anschließend in der trockenen Mehlmischung gründlich panieren.
    
    \step%
    \textbf{Frittieren:} Tofu portionsweise 3--5 Minuten goldbraun fritieren. Auf Küchenpapier abtropfen lassen.
    
    \step%
    \textbf{Sauce zubereiten:} Butter schmelzen, gehackten Knoblauch anschwitzen, Sojasauce und Ahornsirup zugeben und 2--3 Minuten leicht einkochen. Opt. Sriracha einrühren.
    
    \step%
    Tofu in die Sauce geben und vorsichtig schwenken, bis alle Stücke überzogen sind. Sofort servieren.
}

\suggestion[Beilage]{%
    Dazu passt Reis und Brokkoli. Pro Portion rechne ich mit 100 g ungekochtem Reis und 250 g Brokkoli.
}

\end{recipe}
