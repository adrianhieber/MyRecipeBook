\ingredients{
    1 & Großer Brokkoli\index{Gemüse \& Pilze!Brokkoli}\\
    2--3 & Rote Beete (vorgekocht)\index{Gemüse \& Pilze!Rote Beete}\\
    500g & Kichererbsen\index{Hülsenfrüchte!Kichererbsen}\\
    4 EL & Tahin\index{Öle, Saucen \& Würzmittel!Tahin}\\
    1 & Zitrone\index{Obst!Zitronen}\\
    2 & Knoblauchzehen\index{Gemüse \& Pilze!Knoblauch}\\
    2 EL & Balsamico\index{Öle, Saucen \& Würzmittel!Balsamico}\\
    3 EL & Olivenöl\index{Öle, Saucen \& Würzmittel!Olivenöl}\\
    1 TL & Kreuzkümmel\index{Kräuter \& Gewürze!Kreuzkümmel}\\
    2 EL & Petersilie\index{Kräuter \& Gewürze!Petersilie}\\
    & Salz, Pfeffer\index{Kräuter \& Gewürze!Pfeffer}\index{Kräuter \& Gewürze!Salz}
}

\preparation{
    \step%
    Brokkoli in 1cm dicke Scheiben schneiden und in etwas Olivenöl goldbraun anbraten.
    
    \step%
    Für den Hummus Rote Beete, Kichererbsen (mit 2 EL Saft), Tahin, Kreuzkümmel, Knoblauch und Zitronensaft pürieren.

    \step%
    Eiswürfel zugeben und kurz weiter pürieren, bis die Konsistenz cremig ist.
    
    \step%
    Olivenöl mit Balsamico vermischen.\\
    
    \step%
    Petersilie fein hacken.\\
    
    \step%
    Hummus auf einem Teller verstreichen, Brokkoli darauf legen und mit Topping und Petersilie garnieren.
}

\suggestion[Schärfe]{Etwas Chili oder Harissa ins Balsamico-Öl geben.}

\hint{Ein Eiswürfel im Hummus macht ihn besonders luftig und cremig.}

\end{recipe}
