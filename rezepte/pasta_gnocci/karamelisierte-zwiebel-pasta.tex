\begin{recipe}
[
    preparationtime = {\unit[50]{min}},
    portion = {\portion{2}},
    calory = {1040 kcal | 28g Protein | 115g KH | 48g Fett},
    rating = {4},
    source = {Adrian Hieber},
    price = {2,06€}
]
{Karamellisierte Zwiebel Pasta}

\introduction{%
    Süßlich, würzig und cremig durch Kokosmilch und Hefeflocken. 
    Perfekt wenn man Lust auf intensive Zwiebel-Aromen hat.
}

\ingredients{%
    1 EL & Olivenöl\index{Öle, Saucen \& Würzmittel!Olivenöl}\\
    1 EL & Margarine\index{Milchprodukte \& Alternativen!Margarine}\\
    2 & Große Zwiebeln\index{Gemüse \& Pilze!Zwiebeln}\\
    30 g & Getrocknete Tomaten\index{Vorrat \& Konserven!Tomaten, getrocknet}\\
    1 TL & Chili in Öl\index{Öle, Saucen \& Würzmittel!Chiliöl}\\
    \nicefrac{1}{2} TL & Oregano\index{Kräuter \& Gewürze!Oregano}\\
    \nicefrac{1}{2} TL & Basilikum\index{Kräuter \& Gewürze!Basilikum}\\
    \nicefrac{1}{2} TL & Petersilie\index{Kräuter \& Gewürze!Petersilie}\\
    \nicefrac{1}{4} TL & Thymian\index{Kräuter \& Gewürze!Thymian}\\
    \nicefrac{1}{4} TL & Rosmarin\index{Kräuter \& Gewürze!Rosmarin}\\
    \nicefrac{1}{4} TL & Majoran\index{Kräuter \& Gewürze!Majoran}\\
    1 EL & Sojasauce\index{Öle, Saucen \& Würzmittel!Sojasauce}\\
    \nicefrac{1}{2} EL & Balsamico\index{Öle, Saucen \& Würzmittel!Balsamico}\\
    150 ml & Kokosmilch\index{Milchprodukte \& Alternativen!Kokosmilch}\\
    25 g & Hefeflocken\index{Vorrat \& Konserven!Hefeflocken}\\
    4 & Knoblauchzehen\index{Gemüse \& Pilze!Knoblauch}\\
    200 g & Nudeln\index{Getreide, Pasta \& Brot!Pasta}
}

\preparation{%
    \step%
    Zwiebeln in feine Halbringe schneiden und mit Öl und Margarine ca. 35 Minuten karamellisieren, regelmäßig rühren, am Schluss mehr rühren.
    
    \step%
    Knoblauch und getrocknete Tomaten fein schneiden.\\
    
    \step%
    30 Minuten nach Kochbeginn Nudeln in Salzwasser kochen.
    
    \step%
    Restliche Zutaten (außer Kokosmilch und Hefeflocken) zu den Zwiebeln geben und kurz (4 Minuten) anbraten.
    
    \step%
    Kokosmilch und etwas Nudelwasser zugeben und einige Minuten köcheln lassen.
    
    \step%
    Hefeflocken unterrühren.\\
    
    \step%
    Nudeln unterheben und servieren.
}

\hint{
    Wenn es cremiger sein soll, etwas länger köcheln lassen oder leicht pürieren.
}

\end{recipe}
